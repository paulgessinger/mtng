{% macro show_item(item, spec, mode = none, is_new = false) %}
    \item\begin{minipage}[t]{\linewidth}\raggedright
    {%- set truncate = item.is_pr and spec.truncate_pr_titles -%}
    {%- if truncate -%}\mtngprline{ {% endif %}
    {%- if item.is_pr -%}
      {% if mode == "OPEN" %}{% if item.draft %}\prdraft{% else %}\propen{% endif %}{% elif mode == "MERGED" -%}\prmerged{% endif %}
    {%- else -%}
      \iss
    {%- endif -%}
    {%- if item.is_pr and mode == "OPEN" and is_new -%}\prnew{%- endif -%}
    {%- if item.is_pr and item.pr_category_breaking -%}\prbreaking{ {{- spec.pr_category_colors.get("breaking", "alertred") -}} }{%- endif -%}
    {%- if item.draft and mode == "OPEN" -%}\prcategory{Draft}{black!45}
    {%- elif item.is_wip -%}\prwip
    {%- endif -%}
    {%- if item.is_stale -%}\prstale
    {%- endif -%}
    {%- if item.is_pr and spec.show_pr_categories and item.pr_category_label is not none and item.pr_category_color is not none -%}
      \prcategory{
        {{- item.pr_category_label|sanitize -}}
      }{ {{- item.pr_category_color -}} }
    {%- endif -%}
    {%- if truncate -%}
    }{ {{- (item.pr_title_display if item.pr_title_display is not none else item.title)|title_units -}} }{ {{- item.html_url -}} }{
    {%- else -%}
    \textbf{\href{
      {{- item.html_url -}}
    }{\textcolor{black}{
      {{- (item.pr_title_display if item.pr_title_display is not none else item.title)|sanitize -}}
    }}}
    {%- endif %}
    (\href{ {{- item.html_url -}} }{
    {%- if item.is_pr -%}
    {% if spec.forge == "gitlab" %}!{{- item.number -}}{% else %}\prstr{ {{- item.number -}} }{% endif %}
    {%- else -%}
    \issstr{ {{- item.number -}} }
    {%- endif -%}
    }){% if truncate %}}{% endif %} \\
    by {{ user(item.user) }}
    {%- if spec.do_assignee -%}
        ,\space{}
    {%- endif -%}
    {%- if spec.do_assignee -%}
        {%- if item.assignee -%}
            assigned to {{ user(item.assignee) }}
        {%- else -%}
            {%- if spec.no_assignee_attention and mode != "MERGED"-%}
                \emojispace{warning}\textbf{\textcolor{Red}{no assignee}}
            {%- else -%}
                no assignee
            {%- endif -%}
        {%- endif -%}
    {%- endif -%}
    {{- caller() }}
    {%- if item.is_pr and spec.do_reviewers and spec.show_review_summary %}
    {% set status, color = item|review_status -%}
    \allowbreak\hspace{0.3em}\mbox{\scriptsize\textcolor{black!55}{\ensuremath{\cdot}}\,\textcolor{ {{- color -}} }{ {{- status -}} }}
    {%- endif %}
    {%- if item.is_pr and item.additions is defined and item.additions is not none and item.deletions is defined and item.deletions is not none %}
    \allowbreak\hspace{0.4em}\mbox{\scriptsize\textcolor{green!50!black}{+{{ item.additions|human_int }}}\,\textcolor{red!75!black}{\ensuremath{-}{{ item.deletions|human_int }}}}
    {%- endif %}
    {%- if item.is_pr and item.changed_files is defined and item.changed_files is not none %}
    \allowbreak\hspace{0.3em}\mbox{\scriptsize\textcolor{black!55}{\ensuremath{\cdot} {{ item.changed_files|human_int }} {{ "file" if item.changed_files == 1 else "files" }}}}
    {%- endif %}
    \end{minipage}
{% endmacro %}
{% macro user(data) -%}
\href{ {{- data.html_url -}} }{\mtngusername{ {{- ("@" ~ data.login)|title_characters -}} }}
{%- endmacro %}
