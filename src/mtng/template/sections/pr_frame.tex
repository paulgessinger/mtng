{% from "macros.tex" import show_item, show_open_item %}
{% macro render_pr_frame(title, prs, mode, date_prefix) %}
\begin{frame}[allowframebreaks=0.9,t]{ {{- title -}} }

  \setlength{\leftmargini}{\dimexpr\leftmargini+\prlistindent\relax}
  \begin{itemize}
    {% include "list_style.tex" %}
    {% if mode == "OPEN" %}
    {% for wip in ["false", "true"] %}
    {% for pr in prs|selectattr("is_wip", wip)|sort(reverse=True, attribute="updated_at") %}
    {{- show_open_item(pr, repo["spec"], since, now) }}
    {%- endfor %}
    {%- endfor %}
    {% else %}
    {% for pr in prs|sort(attribute="closed_at") %}
    {%- call show_item(pr, repo["spec"], mode=mode, is_new=since is not none and since <= pr.created_at <= now) -%}
    , {{ date_prefix }} {{ pr.closed_at.strftime('%Y-%m-%d') }}
    {%- endcall %}
    {%- endfor %}
    {% endif %}
  {# Remove trailing list glue/penalty so it cannot create an empty continuation. #}\end{itemize}\par\unskip\unpenalty

\end{frame}
{% endmacro %}
